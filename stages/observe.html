Procedures used

calculate-period

Divide the elapsed time by the number of full swings.

Inputs: elapsed, swings

Method text

start stage Observe by assistant from state [
  set information to [
    set large-swing to [
      set elapsed to 20.1 with uncertainty [
        set declared to [set reaction-time to [set type to "epistemic" set estimate to 0.2 set unit to "s"]]
      ]
      set swing-count to 10
      set unit to "s"
    ]
    set small-swing to [
      set elapsed to 20.0 with uncertainty [
        set declared to [set reaction-time to [set type to "epistemic" set estimate to 0.2 set unit to "s"]]
      ]
      set swing-count to 10
      set unit to "s"
    ]
  ]
  set environment to [
    set pendulum to [set length to 1 set unit to "m"]
    set timer to "hand stopwatch"
  ]
  set factors to [set reaction-time to unknown]
]
  about "Time ten full swings at a large and at a small swing, and derive the
  period of each."
  do calculate-period with [
    elapsed at elapsed of information.large-swing
    swings at swing-count of information.large-swing
  ] into period of information.large-swing
  do calculate-period with [
    elapsed at elapsed of information.small-swing
    swings at swing-count of information.small-swing
  ] into period of information.small-swing
end stage with state [
  set information to [
    set large-swing to [
      set elapsed to 20.1 with uncertainty [
        set declared to [set reaction-time to [set type to "epistemic" set estimate to 0.2 set unit to "s"]]
      ]
      set swing-count to 10
      set unit to "s"
      set period to unknown
    ]
    set small-swing to [
      set elapsed to 20.0 with uncertainty [
        set declared to [set reaction-time to [set type to "epistemic" set estimate to 0.2 set unit to "s"]]
      ]
      set swing-count to 10
      set unit to "s"
      set period to unknown
    ]
  ]
  set environment to [
    set pendulum to [set length to 1 set unit to "m"]
    set timer to "hand stopwatch"
  ]
  set factors to [set reaction-time to unknown]
]
